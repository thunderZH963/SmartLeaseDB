cmake_minimum_required(VERSION 3.16)
project(SmartLeaseDB VERSION 0.1.0 LANGUAGES CXX)

option(SMARTLEASE_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
find_package(Threads REQUIRED)
add_library(smartlease
  src/controller.cpp src/broker.cpp src/enforcer.cpp src/execution.cpp)
target_include_directories(smartlease PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(smartlease PUBLIC cxx_std_17)
set_target_properties(smartlease PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(smartlease PUBLIC Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(smartlease PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(SMARTLEASE_SANITIZE)
    target_compile_options(smartlease PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(smartlease PUBLIC -fsanitize=address,undefined)
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  foreach(suite controller enforcer execution broker integration)
    add_executable(${suite}_test tests/${suite}_test.cpp)
    target_link_libraries(${suite}_test PRIVATE smartlease)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
      target_compile_options(${suite}_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME ${suite} COMMAND ${suite}_test)
    set_tests_properties(${suite} PROPERTIES TIMEOUT 60)
  endforeach()
endif()
add_executable(smartlease-demo examples/reclaim_demo.cpp)
target_link_libraries(smartlease-demo PRIVATE smartlease)
